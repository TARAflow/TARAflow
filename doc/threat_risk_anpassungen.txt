type VerificationMethod =
  | 'unit_test'           // funktionale Korrektheit
  | 'integration_test'    // Zusammenspiel von Komponenten
  | 'negative_test'       // ungültige Inputs, Boundary Cases
  | 'fuzz_test'           // randomisierter Input
  | 'load_test'           // Verhalten unter Normallast
  | 'stress_test'         // Verhalten über dem Limit / Ressourcenerschöpfung
  | 'penetration_test'    // freies adversariales Angriffsszenario
  | 'code_review'         // manuelle Codeanalyse
  | 'static_analysis'     // automatisierte Codeanalyse (SAST)
  | 'audit'               // formale Überprüfung durch Dritte
  | 'formal_analysis'     // mathematischer Beweis
  | 'field_test';         // Validierung im realen System


type VerificationStatus =
  | 'pending'     // noch nicht durchgeführt
  | 'passed'      // erfolgreich verifiziert
  | 'failed'      // Verifikation fehlgeschlagen — Mitigation wirkt nicht
  | 'waived';     // bewusst nicht verifiziert — Begründung Pflicht
  

// Threat Tab — bei der Mitigation
interface MitigationEntry {
  description: string;
  method: VerificationMethod;
}

// Risk Tab — nach dem Treatment Decision
interface VerificationEntry {
  status: VerificationStatus;
  ticketRef?: string;       // z.B. "JIRA-1234" oder "ADO-567"
  date?: string;
  owner?: string;
  rationale?: string;       // Pflicht wenn 'waived' oder 'failed'
}

T-ID | STRIDE | Source | Severity | Safety | T-Description | Attack Description | Actor* | Mitigation | Verification Method | Actions
* optional
Source — derived / manual / relation: Analyst sieht sofort ob der Threat automatisch generiert wurde oder manuell erfasst — wichtig für Audit

Severity — CRITICAL / HIGH / MEDIUM / LOW: 
MAX-Wert aus CIANAAA der betroffenen Assets Safety Override bereits eingerechnet → Analyst sieht sofort Priorität ohne Risk-Tab zu öffnen

Safety — direct / indirect / none: Schnelles Signal ob Safety Override greift → diese Threats zuerst bearbeiten

Nur die mitigation_type Chips anzeigen — kein Text, kein ↳.
Bei Hover auf einen Chip erscheint ein Tooltip mit suggestion + notes.
Bei Klick auf die Zelle öffnet sich der Dialog mit allen Details.
Vorteile:
Chips        → platzsparend, auf einen Blick erkennbar
Tooltip      → Details ohne Dialog zu öffnen
Dialog       → vollständige Erfassung/Bearbeitung
